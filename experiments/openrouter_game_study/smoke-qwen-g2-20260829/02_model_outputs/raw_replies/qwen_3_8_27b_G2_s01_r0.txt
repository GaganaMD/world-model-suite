class GeneratedPOSG(AbstractPOSG):
    @property
    def players(self):
        return ["player_1", "player_2"]

    def initial_state(self):
        return "s0"

    def legal_actions(self, state, player):
        # Each player chooses a die face 1-6
        return ["1", "2", "3", "4", "5", "6"]

    def transition(self, state, joint_action):
        # joint_action is a dict: {"player_1": "1".."6", "player_2": "1".."6"}
        # The game is deterministic given the joint action (both choose a face)
        # Terminal state encodes the outcome
        p1 = joint_action["player_1"]
        p2 = joint_action["player_2"]
        next_state = f"terminal_{p1}_{p2}"
        return {next_state: 1.0}

    def observation(self, state, joint_action, next_state, player):
        # Each player observes their own roll and the opponent's roll
        p1 = joint_action["player_1"]
        p2 = joint_action["player_2"]
        if player == "player_1":
            return f"my_roll={p1}, opp_roll={p2}"
        else:
            return f"my_roll={p2}, opp_roll={p1}"

    def reward(self, state, joint_action, next_state, player):
        p1 = int(joint_action["player_1"])
        p2 = int(joint_action["player_2"])
        if player == "player_1":
            if p1 > p2:
                return 1.0
            elif p1 < p2:
                return -1.0
            else:
                return 0.0
        else:
            if p2 > p1:
                return 1.0
            elif p2 < p1:
                return -1.0
            else:
                return 0.0

    def is_terminal(self, state):
        return state.startswith("terminal_")
