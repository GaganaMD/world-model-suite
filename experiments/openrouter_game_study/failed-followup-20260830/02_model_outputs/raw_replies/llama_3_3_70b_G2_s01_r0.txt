class GeneratedPOSG(AbstractPOSG):
    @property
    def players(self): return ["player_1", "player_2"]
    def initial_state(self): return "start"
    def legal_actions(self, state, player): 
        if state == "start": 
            return ["roll"]
        elif state.startswith("result_"): 
            return ["reset"]
        else: 
            return []
    def transition(self, state, joint_action): 
        if state == "start" and joint_action["player_1"] == "roll" and joint_action["player_2"] == "roll":
            outcomes = {}
            for roll_1 in range(1, 7):
                for roll_2 in range(1, 7):
                    outcome = f"result_{roll_1}_{roll_2}"
                    if outcome not in outcomes:
                        outcomes[outcome] = 0.0
                    outcomes[outcome] += 1.0 / 36.0
            return outcomes
        elif state.startswith("result_") and joint_action["player_1"] == "reset" and joint_action["player_2"] == "reset":
            return {"start": 1.0}
        else:
            return {}
    def observation(self, state, joint_action, next_state, player): 
        if state == "start" and joint_action["player_1"] == "roll" and joint_action["player_2"] == "roll":
            roll_1, roll_2 = map(int, next_state.split("_")[1:])
            if player == "player_1":
                return f"you_rolled_{roll_1}_opponent_rolled_{roll_2}"
            else:
                return f"you_rolled_{roll_2}_opponent_rolled_{roll_1}"
        elif state.startswith("result_") and joint_action["player_1"] == "reset" and joint_action["player_2"] == "reset":
            return "game_reset"
        else:
            return ""
    def reward(self, state, joint_action, next_state, player): 
        if state == "start" and joint_action["player_1"] == "roll" and joint_action["player_2"] == "roll":
            roll_1, roll_2 = map(int, next_state.split("_")[1:])
            if player == "player_1":
                if roll_1 > roll_2:
                    return 1.0
                elif roll_1 < roll_2:
                    return -1.0
                else:
                    return 0.0
            else:
                if roll_1 < roll_2:
                    return 1.0
                elif roll_1 > roll_2:
                    return -1.0
                else:
                    return 0.0
        elif state.startswith("result_") and joint_action["player_1"] == "reset" and joint_action["player_2"] == "reset":
            return 0.0
        else:
            return 0.0
    def is_terminal(self, state): 
        return state.startswith("result_")
