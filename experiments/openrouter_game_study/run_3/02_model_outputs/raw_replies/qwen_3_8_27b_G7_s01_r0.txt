class GeneratedPOSG(AbstractPOSG):
    @property
    def players(self):
        return ["player_1", "player_2"]

    def initial_state(self):
        return "s0"

    def legal_actions(self, state, player):
        if state == "s0":
            return ["silent", "testify"]
        return ["noop"]

    def transition(self, state, joint_action):
        if state == "s0":
            a1 = joint_action.get("player_1", "silent")
            a2 = joint_action.get("player_2", "silent")
            if a1 == "silent" and a2 == "silent":
                return {"terminal_both_silent": 1.0}
            elif a1 == "testify" and a2 == "testify":
                return {"terminal_both_testify": 1.0}
            elif a1 == "testify" and a2 == "silent":
                return {"terminal_p1_testify_p2_silent": 1.0}
            else:
                return {"terminal_p1_silent_p2_testify": 1.0}
        return {"terminal": 1.0}

    def observation(self, state, joint_action, next_state, player):
        if state == "s0":
            a1 = joint_action.get("player_1", "silent")
            a2 = joint_action.get("player_2", "silent")
            if player == "player_1":
                return f"state={state};my_action={a1};next_state={next_state}"
            else:
                return f"state={state};my_action={a2};next_state={next_state}"
        return f"state={state};next_state={next_state}"

    def reward(self, state, joint_action, next_state, player):
        if state == "s0":
            a1 = joint_action.get("player_1", "silent")
            a2 = joint_action.get("player_2", "silent")
            if player == "player_1":
                if a1 == "silent" and a2 == "silent":
                    return -1.0
                elif a1 == "testify" and a2 == "testify":
                    return -3.0
                elif a1 == "testify" and a2 == "silent":
                    return 0.0
                else:
                    return -5.0
            else:
                if a1 == "silent" and a2 == "silent":
                    return -1.0
                elif a1 == "testify" and a2 == "testify":
                    return -3.0
                elif a1 == "silent" and a2 == "testify":
                    return 0.0
                else:
                    return -5.0
        return 0.0

    def is_terminal(self, state):
        return state != "s0"
