class GeneratedPOSG(AbstractPOSG):
    @property
    def players(self):
        return ["player_1", "player_2"]

    def initial_state(self):
        return "s0"

    def legal_actions(self, state, player):
        if state == "terminal":
            return []
        return ["0_3", "1_2", "2_1", "3_0"]

    def transition(self, state, joint_action):
        if state == "terminal":
            return {"terminal": 1.0}
        p1_action = joint_action["player_1"]
        p2_action = joint_action["player_2"]
        p1_battlefield1 = int(p1_action.split("_")[0])
        p1_battlefield2 = int(p1_action.split("_")[1])
        p2_battlefield1 = int(p2_action.split("_")[0])
        p2_battlefield2 = int(p2_action.split("_")[1])
        p1_wins_b1 = 1 if p1_battlefield1 > p2_battlefield1 else 0
        p1_wins_b2 = 1 if p1_battlefield2 > p2_battlefield2 else 0
        p1_total_wins = p1_wins_b1 + p1_wins_b2
        p2_total_wins = 2 - p1_total_wins
        if p1_total_wins > p2_total_wins:
            outcome = "p1_win"
        elif p2_total_wins > p1_total_wins:
            outcome = "p2_win"
        else:
            outcome = "draw"
        return {outcome: 1.0}

    def observation(self, state, joint_action, next_state, player):
        if state == "terminal":
            return "terminal_obs"
        p1_action = joint_action["player_1"]
        p2_action = joint_action["player_2"]
        if player == "player_1":
            return f"p1_played_{p1_action}_p2_played_{p2_action}"
        else:
            return f"p2_played_{p2_action}_p1_played_{p1_action}"

    def reward(self, state, joint_action, next_state, player):
        if state == "terminal":
            return 0.0
        p1_action = joint_action["player_1"]
        p2_action = joint_action["player_2"]
        p1_battlefield1 = int(p1_action.split("_")[0])
        p1_battlefield2 = int(p1_action.split("_")[1])
        p2_battlefield1 = int(p2_action.split("_")[0])
        p2_battlefield2 = int(p2_action.split("_")[1])
        p1_wins_b1 = 1 if p1_battlefield1 > p2_battlefield1 else 0
        p1_wins_b2 = 1 if p1_battlefield2 > p2_battlefield2 else 0
        p1_total_wins = p1_wins_b1 + p1_wins_b2
        p2_total_wins = 2 - p1_total_wins
        if p1_total_wins > p2_total_wins:
            outcome = "p1_win"
        elif p2_total_wins > p1_total_wins:
            outcome = "p2_win"
        else:
            outcome = "draw"
        if player == "player_1":
            if outcome == "p1_win":
                return 1.0
            elif outcome == "p2_win":
                return -1.0
            else:
                return 0.0
        else:
            if outcome == "p2_win":
                return 1.0
            elif outcome == "p1_win":
                return -1.0
            else:
                return 0.0

    def is_terminal(self, state):
        return state == "terminal"
