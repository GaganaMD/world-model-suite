class GeneratedPOSG(AbstractPOSG):

    @property
    def players(self):
        return ["player_1", "player_2"]

    def initial_state(self): 
        return "deploy"

    def legal_actions(self, state, player):
        if state == "deploy":
            return ["3-0", "2-1", "1-2", "0-3"]
        return ["noop"]

    def _parse(self, a):
        parts = a.split("-")
        return int(parts[0]), int(parts[1])

    def _outcome(self, a1, a2):
        x1, y1 = self._parse(a1)
        x2, y2 = self._parse(a2)
        w1 = 0
        w2 = 0
        if x1 > x2:
            w1 += 1
        elif x2 > x1:
            w2 += 1
        if y1 > y2:
            w1 += 1
        elif y2 > y1:
            w2 += 1
        if w1 > w2:
            return "win1"
        if w2 > w1:
            return "win2"
        return "draw"

    def transition(self, state, joint_action):
        if state == "deploy":
            a1 = joint_action["player_1"]
            a2 = joint_action["player_2"]
            res = self._outcome(a1, a2)
            return {"end_" + res + "_" + a1 + "_" + a2: 1.0}
        return {"end_draw_0-3_0-3": 1.0}

    def observation(self, state, joint_action, next_state, player):
        if state != "deploy":
            return "terminal"
        a1 = joint_action["player_1"]
        a2 = joint_action["player_2"]
        res = self._outcome(a1, a2)
        own = a1 if player == "player_1" else a2
        opp = a2 if player == "player_1" else a1
        return f"own={own}|opp={opp}|result={res}"

    def reward(self, state, joint_action, next_state, player):
        if state != "deploy":
            return 0.0
        res = self._outcome(joint_action["player_1"], joint_action["player_2"])
        if res == "draw":
            return 0.0
        if res == "win1":
            return 1.0 if player == "player_1" else -1.0
        return -1.0 if player == "player_1" else 1.0

    def is_terminal(self, state):
        return state.startswith("end_")
