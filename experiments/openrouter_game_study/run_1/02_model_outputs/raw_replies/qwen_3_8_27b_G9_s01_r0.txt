class GeneratedPOSG(AbstractPOSG):
    @property
    def players(self):
        return ["player_1", "player_2"]

    def initial_state(self):
        return "s0"

    def legal_actions(self, state, player):
        # Each commander allocates 3 soldiers between two battlefields.
        # Action is a string encoding the split: "a-b" where a+b=3
        # a = soldiers to battlefield 1, b = soldiers to battlefield 2
        return ["0-3", "1-2", "2-1", "3-0"]

    def transition(self, state, joint_action):
        # This is a one-shot game. After the joint action, we go to a terminal state.
        # The terminal state encodes the outcome for observation/reward purposes.
        # But per contract, transition returns dict[str, float] of next states.
        # We need to determine the outcome to create a meaningful terminal state.
        
        p1_action = joint_action["player_1"]
        p2_action = joint_action["player_2"]
        
        # Parse actions
        p1_b1, p1_b2 = map(int, p1_action.split("-"))
        p2_b1, p2_b2 = map(int, p2_action.split("-"))
        
        # Determine battlefield winners
        # Battlefield 1: higher soldiers wins
        if p1_b1 > p2_b1:
            b1_winner = "player_1"
        elif p2_b1 > p1_b1:
            b1_winner = "player_2"
        else:
            b1_winner = "draw"
        
        # Battlefield 2: higher soldiers wins
        if p1_b2 > p2_b2:
            b2_winner = "player_1"
        elif p2_b2 > p1_b2:
            b2_winner = "player_2"
        else:
            b2_winner = "draw"
        
        # Determine overall winner
        p1_wins = 0
        p2_wins = 0
        if b1_winner == "player_1":
            p1_wins += 1
        elif b1_winner == "player_2":
            p2_wins += 1
        
        if b2_winner == "player_1":
            p1_wins += 1
        elif b2_winner == "player_2":
            p2_wins += 1
        
        if p1_wins > p2_wins:
            overall = "player_1_wins"
        elif p2_wins > p1_wins:
            overall = "player_2_wins"
        else:
            overall = "draw"
        
        terminal_state = f"terminal_{overall}"
        return {terminal_state: 1.0}

    def observation(self, state, joint_action, next_state, player):
        # In this game, each player only observes their own action and the outcome.
        # They do NOT observe the other player's action (simultaneous deployment).
        # The observation should encode what the player knows.
        
        p1_action = joint_action["player_1"]
        p2_action = joint_action["player_2"]
        
        # Parse to determine outcome
        p1_b1, p1_b2 = map(int, p1_action.split("-"))
        p2_b1, p2_b2 = map(int, p2_action.split("-"))
        
        if p1_b1 > p2_b1:
            b1_winner = "player_1"
        elif p2_b1 > p1_b1:
            b1_winner = "player_2"
        else:
            b1_winner = "draw"
        
        if p1_b2 > p2_b2:
            b2_winner = "player_1"
        elif p2_b2 > p1_b2:
            b2_winner = "player_2"
        else:
            b2_winner = "draw"
        
        p1_wins = 0
        p2_wins = 0
        if b1_winner == "player_1":
            p1_wins += 1
        elif b1_winner == "player_2":
            p2_wins += 1
        
        if b2_winner == "player_1":
            p1_wins += 1
        elif b2_winner == "player_2":
            p2_wins += 1
        
        if p1_wins > p2_wins:
            overall = "player_1_wins"
        elif p2_wins > p1_wins:
            overall = "player_2_wins"
        else:
            overall = "draw"
        
        # Each player observes: their own action, the battlefield outcomes, and overall result
        # They do NOT observe the opponent's exact allocation
        own_action = p1_action if player == "player_1" else p2_action
        return f"own={own_action};b1={b1_winner};b2={b2_winner};overall={overall}"

    def reward(self, state, joint_action, next_state, player):
        # Reward: +1 for winning, 0 for draw, -1 for losing
        p1_action = joint_action["player_1"]
        p2_action = joint_action["player_2"]
        
        p1_b1, p1_b2 = map(int, p1_action.split("-"))
        p2_b1, p2_b2 = map(int, p2_action.split("-"))
        
        if p1_b1 > p2_b1:
            b1_winner = "player_1"
        elif p2_b1 > p1_b1:
            b1_winner = "player_2"
        else:
            b1_winner = "draw"
        
        if p1_b2 > p2_b2:
            b2_winner = "player_1"
        elif p2_b2 > p1_b2:
            b2_winner = "player_2"
        else:
            b2_winner = "draw"
        
        p1_wins = 0
        p2_wins = 0
        if b1_winner == "player_1":
            p1_wins += 1
        elif b1_winner == "player_2":
            p2_wins += 1
        
        if b2_winner == "player_1":
            p1_wins += 1
        elif b2_winner == "player_2":
            p2_wins += 1
        
        if p1_wins > p2_wins:
            overall = "player_1_wins"
        elif p2_wins > p1_wins:
            overall = "player_2_wins"
        else:
            overall = "draw"
        
        if player == "player_1":
            if overall == "player_1_wins":
                return 1.0
            elif overall == "player_2_wins":
                return -1.0
            else:
                return 0.0
        else:
            if overall == "player_2_wins":
                return 1.0
            elif overall == "player_1_wins":
                return -1.0
            else:
                return 0.0

    def is_terminal(self, state):
        return state.startswith("terminal_")
